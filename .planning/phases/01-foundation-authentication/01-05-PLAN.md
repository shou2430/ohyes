2. On mount, read `token` from URL search params via `useSearchParams()`.
3. If `token` exists:
   - Store it in `localStorage` with key `"ohyes_token"`.
   - Navigate to `/dashboard` with `{ replace: true }` to prevent back-button loops.
4. If `token` is missing (or if an `error` param is present):
   - Navigate to `/` with `{ replace: true }`.
5. Render a minimal loading indicator (e.g., a centered "Signing in..." text with Tailwind classes) while the redirect happens.

```jsx
import { useEffect } from "react";
import { useSearchParams, useNavigate } from "react-router";

export default function AuthCallbackPage() {
  const [searchParams] = useSearchParams();
  const navigate = useNavigate();

  useEffect(() => {
    const token = searchParams.get("token");
    const error = searchParams.get("error");

    if (token) {
      localStorage.setItem("ohyes_token", token);
      navigate("/dashboard", { replace: true });
    } else {
      navigate("/", { replace: true });
    }
  }, [searchParams, navigate]);

  return (
    <div className="flex min-h-screen items-center justify-center bg-cream">
      <p className="text-lg text-gray-600">Signing in...</p>
    </div>
  );
}
```

Rewrite `frontend/src/context/AuthContext.jsx` to manage the JWT token from localStorage:

1. **Token key constant:** `const TOKEN_KEY = "ohyes_token";`

2. **Helper function** to build auth headers:
   ```jsx

     const token = localStorage.getItem(TOKEN_KEY);
     if (!token) return {};
     return { Authorization: `Bearer ${token}` };
   }
   ```

3. **checkAuth (in useEffect on mount):**
   - Read token from `localStorage.getItem(TOKEN_KEY)`.
   - If no token exists, set `user` to `null` and `loading` to `false` immediately (skip the fetch).
   - If token exists, call `fetch("/api/auth/me", { headers: authHeaders() })`.
   - If response is OK (200), parse JSON and set `user`.
   - If response is 401 or 403, the token is expired/invalid: call `localStorage.removeItem(TOKEN_KEY)` and set `user` to `null`.
   - On network error, set `user` to `null` (do NOT clear token — may be a transient failure).

4. **logout function:**
   - Call `fetch("/api/auth/logout", { method: "POST", headers: authHeaders() })`.
   - In the `finally` block: call `localStorage.removeItem(TOKEN_KEY)` and set `user` to `null`.

5. **Remove** the `credentials: "same-origin"` option from both fetch calls — it is not needed with Bearer token auth.

6. **Add a `token` value to context** so components can access it if needed:
   - Add `const [token, setToken] = useState(() => localStorage.getItem(TOKEN_KEY));`
   - Expose `token` in the context value (optional but useful for future API calls).

Full replacement for `AuthContext.jsx`:

```jsx
import { createContext, useContext, useEffect, useState } from "react";

const AuthContext = createContext(null);

const TOKEN_KEY = "ohyes_token";

function authHeaders() {
  const token = localStorage.getItem(TOKEN_KEY);
  if (!token) return {};
  return { Authorization: `Bearer ${token}` };
}

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function checkAuth() {
      const token = localStorage.getItem(TOKEN_KEY);
      if (!token) {
        setLoading(false);
        return;
      }

      try {
        const res = await fetch("/api/auth/me", {
          headers: authHeaders(),
        });
        if (res.ok) {
          const data = await res.json();
          setUser(data);
        } else {
          localStorage.removeItem(TOKEN_KEY);
          setUser(null);
        }
      } catch {
        setUser(null);
      } finally {
        setLoading(false);
      }
    }
    checkAuth();
  }, []);

  const logout = async () => {
    try {
      await fetch("/api/auth/logout", {
        method: "POST",
        headers: authHeaders(),
      });
    } finally {
      localStorage.removeItem(TOKEN_KEY);
      setUser(null);
    }
  };

  return (
    <AuthContext.Provider value={{ user, loading, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error("useAuth must be used within an AuthProvider");
  }
  return context;
}
```
